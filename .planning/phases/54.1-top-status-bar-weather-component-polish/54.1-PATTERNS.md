```qml
    // Temperature parsing & reactive telemetry (D-54.1-14, D-54.1-17, D-54.1-28)
    readonly property string formattedTemp: {
        const raw = Weather.current?.tempC;
        if (raw === undefined || raw === null || raw === "" || raw === "--") {
            return (root.useShortenedForm === 2 || root.vertical) ? "--°" : "--°C";
        }
        const num = Math.round(Number(raw));

            return (root.useShortenedForm === 2 || root.vertical) ? "--°" : "--°C";
        }
        return num + ((root.useShortenedForm === 2 || root.vertical) ? "°" : "°C");
    }
```

#### G. Imminent Rain Badge & Severe Weather Alert Precedence
Scans 3-hour window for rain probability > 50%; severe alert suppresses rain badge and suppresses rain in vertical mode ([`D-54.1-15`](file:///home/pera/github_repo/.dotfiles/.planning/phases/54.1-top-status-bar-weather-component-polish/54.1-CONTEXT.md#L62), [`D-54.1-16`](file:///home/pera/github_repo/.dotfiles/.planning/phases/54.1-top-status-bar-weather-component-polish/54.1-CONTEXT.md#L63), [`D-54.1-22`](file:///home/pera/github_repo/.dotfiles/.planning/phases/54.1-top-status-bar-weather-component-polish/54.1-CONTEXT.md#L71), [`D-54.1-25`](file:///home/pera/github_repo/.dotfiles/.planning/phases/54.1-top-status-bar-weather-component-polish/54.1-CONTEXT.md#L76)):
```qml
    readonly property int imminentRainChance: {
        if (!Weather.hourly || Weather.hourly.length === 0) return 0;
        let maxChance = 0;
        const slots = Math.min(3, Weather.hourly.length);
        for (let i = 0; i < slots; ++i) {
            const chance = parseInt(Weather.hourly[i]?.chanceofrain ?? "0", 10);
            if (!isNaN(chance) && chance > maxChance) {
                maxChance = chance;
            }
        }
        return maxChance;
    }

    readonly property bool imminentRain: imminentRainChance > 50

    readonly property bool hasSevereAlert: (Weather.alerts !== undefined && Weather.alerts !== null && Weather.alerts.length > 0)
    readonly property var activeAlert: hasSevereAlert ? Weather.alerts[0] : null
    
    // Dim severe alert icon when cached telemetry is stale or offline (D-54.1-22)
    readonly property color alertColor: (Weather.isStale || Weather.isOffline) ?
        Appearance.m3colors.m3onSurfaceVariant :
        (activeAlert?.color ?? (activeAlert?.severity ? WeatherGlyphs.getAlertColor(activeAlert.severity) : Appearance.m3colors.m3error))

    // Precedence: severe alert suppresses rain badge; suppressed in Tier 2 and vertical bar mode (D-54.1-16, D-54.1-17, D-54.1-25)
    readonly property bool showRainBadge: imminentRain && !hasSevereAlert && (root.useShortenedForm < 2) && !root.vertical
```

## Anti-Patterns to Avoid

| Anti-Pattern | Bad Example | Correct Pattern | Reason |
|---|---|---|---|
| **Double Padding** | `implicitWidth: contentLayout.implicitWidth + 10 * 2` | `implicitWidth: contentLayout.implicitWidth` | `BarGroup` already supplies 5px padding on each side. Adding 20px extra inside causes wide, bloated pills that break Center Zone rhythm. |
| **Nested Surface Containers** | Wrapping `MouseArea` in another `BarGroup` or heavy `Rectangle` container | Subtle `Rectangle` highlight with negative horizontal margins (`leftMargin: -5`, `rightMargin: -5`) aligning with `BarGroup.background` | Avoids competing borders, rounded corners, or double drop shadows. |
| **Click-Based Popup Toggles** | `onClicked: popup.active = !popup.active` or right-click `Weather.getData()` | `onPressed: event => event.accepted = true`<br>`onClicked: event => event.accepted = true`<br>`hoverOpenDelayMs: 300` | User explicitly mandated 100% hover-driven trigger with 300ms delay; clicks must be completely inert. |
| **Actionable Pointer Cursor** | `cursorShape: Qt.PointingHandCursor` | `cursorShape: Qt.ArrowCursor` | Pill is passive telemetry; pointing hand cursor confuses users into clicking. |
| **Color Clashing on Hover** | Dynamic colored hover tint based on alert severity | Neutral `Appearance.colors.colLayer1Hover` at all times | Preserves contrast for severe alert warning icon and imminent rain badge ([`D-54.1-07`](file:///home/pera/github_repo/.dotfiles/.planning/phases/54.1-top-status-bar-weather-component-polish/54.1-CONTEXT.md#L50)). |
| **Rain Badge in Vertical Bar** | Showing rain percentage in vertical mode | `showRainBadge: imminentRain && ... && !root.vertical` | Squeezing rain percentage into a 46px vertical bar balloons pill height and breaks vertical layout aesthetics. |
| **Direct Vendor Edits** | Editing `vendor/dots-hyprland/.../WeatherBar.qml` | Authoring in `restow/quickshell/.../WeatherBar.qml` | Breaks upstream git hygiene; strictly prohibited. |
