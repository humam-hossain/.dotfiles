##### Standardized Dual-Tier Canvas 2D Setup

```qml
    Item {
        id: graphArea
        Layout.fillWidth: true
        Layout.fillHeight: true

        Canvas {
            id: graphCanvas
            anchors.fill: parent
            renderStrategy: Canvas.Immediate
            renderTarget: Canvas.FramebufferObject

            onPaint: {
                const ctx = getContext("2d");
                ctx.clearRect(0, 0, width, height);

                const hours = root.validHours;
                if (!hours || hours.length < 2) return;

                const plotLeft = 14;
                const plotRight = width - 14;
                const plotW = Math.max(10, plotRight - plotLeft);
                const n = hours.length;
                const slotW = plotW / n;

                const plotTop = 20;
                const plotUpperBottom = height * 0.60;
                const plotLowerTop = height * 0.66;
                const plotLowerBottom = height - 28;

                // 1. Nocturnal background shading (D-54.3-24)
                ctx.save();
                ctx.fillStyle = Appearance.colors.colLayer1 || "#1e1e2e";
                ctx.globalAlpha = 0.12;
                for (let i = 0; i < n; ++i) {
                    if (!hours[i].isDay) {
                        ctx.fillRect(plotLeft + i * slotW, 0, slotW, height);
                    }
                }
                ctx.restore();

                // 2. Midnight 00:00 divider with compact date label (D-54.3-25)
                for (let i = 0; i < n; ++i) {
                    if (hours[i].isDayBoundary || hours[i].hourNum === 0) {
                        const midX = plotLeft + i * slotW;
                        ctx.save();
                        ctx.strokeStyle = Appearance.colors.colOutlineVariant || "#444";
                        ctx.setLineDash([2, 3]);
                        ctx.beginPath();
                        ctx.moveTo(midX, plotTop);
                        ctx.lineTo(midX, plotLowerBottom);
                        ctx.stroke();

                        ctx.setLineDash([]);
                        ctx.font = "9px sans-serif";
                        ctx.fillStyle = Appearance.colors.colOnSurfaceVariant || "#888";
                        ctx.textAlign = "center";
                        const dateChip = root.formatDateChip(hours[i].dateStr);
                        ctx.fillText(dateChip, midX, plotTop - 4);
                        ctx.restore();
                    }
                }

                // 3. Center NOW Hairline & Pill Badge (Slot 12)
                if (root.isTimeline25) {
                    const nowX = plotLeft + (12 + 0.5) * slotW;
                    ctx.save();
                    ctx.strokeStyle = Appearance.colors.colPrimary || "#4fc3f7";
                    ctx.lineWidth = 1.5;
                    ctx.setLineDash([3, 3]);
                    ctx.beginPath();
                    ctx.moveTo(nowX, plotTop);
                    ctx.lineTo(nowX, plotLowerBottom);
                    ctx.stroke();

                    ctx.setLineDash([]);
                    const badgeW = 32;
                    const badgeH = 14;
                    const badgeX = nowX - badgeW / 2;
                    const badgeY = plotTop - 14;
                    ctx.fillStyle = Appearance.colors.colPrimary || "#4fc3f7";
                    ctx.beginPath();
                    ctx.rect(badgeX, badgeY, badgeW, badgeH);

                    ctx.textAlign = "center";
                    ctx.textBaseline = "middle";

                    ctx.restore();
```

```javascript
function getMetricY(metricKey, slotIndex, plotTop, plotBottom) {
    const hours = root.validHours;
    if (!hours || slotIndex < 0 || slotIndex >= hours.length) return plotBottom;
    const h = hours[slotIndex];
    if (root.isTimeline25 && h.isHistoricalUnavailable) return plotBottom;

    const plotHeight = plotBottom - plotTop;

    switch (metricKey) {
        case "temp": {
            const bounds = root.calcTempBounds();
            const range = Math.max(1, bounds.max - bounds.min);
            const val = (h.thermal && h.thermal.tempC !== null) ? h.thermal.tempC : (Number(h.tempC || 20));
            return plotBottom - ((val - bounds.min) / range) * plotHeight;
        }
        case "rain": {
            const prob = (h.moisture ? h.moisture.rainProb : (parseInt(h.chanceofrain || 0, 10))) || 0;
            return plotBottom - (Math.max(0, Math.min(100, prob)) / 100.0) * plotHeight;
        }
        case "wind": {
            const maxSpeed = root.calcWindMax();
            const val = (h.wind ? h.wind.speedKmph : (parseInt(h.windspeedKmph || 0, 10))) || 0;
            return plotBottom - (val / maxSpeed) * plotHeight;
        }
        case "aqi": {
            const epa = (h.aqi ? h.aqi.epaIndex : (parseInt(h.air_quality ? h.air_quality["us-epa-index"] : 1, 10))) || 1;
            const clamped = Math.max(1, Math.min(6, epa));
            return plotBottom - ((clamped - 1) / 5.0) * plotHeight;
        }
        case "baro": {
            const bounds = root.calcBaroBounds();
            const range = Math.max(4, bounds.max - bounds.min);
            const val = (h.atmosphere ? h.atmosphere.pressureHpa : (parseInt(h.pressure || 1013, 10))) || 1013;
            return plotBottom - ((val - bounds.min) / range) * plotHeight;
        }
        case "uv": {
            const maxUV = root.calcUvMax();
            const val = (h.solar ? h.solar.uvIndex : (parseInt(h.uvIndex || 0, 10))) || 0;
            return plotBottom - (val / maxUV) * plotHeight;
        }
        default:
            return plotBottom;
    }
}
```

##### Fritsch-Carlson Monotone Cubic Spline Interpolation
Extract control points preserving monotonic progression without local overshoot/undershoot:
```javascript
function computeMonotoneSplineControlPoints(points) {
    const n = points.length;
    if (n < 2) return [];

    const dx = [];
    const dy = [];
    const m = [];
    for (let i = 0; i < n - 1; ++i) {
        const dxi = points[i + 1].x - points[i].x;
        const dyi = points[i + 1].y - points[i].y;
        dx.push(dxi);
        dy.push(dyi);
        m.push(dxi === 0 ? 0 : dyi / dxi);
    }

    const tangents = new Array(n);
    tangents[0] = m[0];
    tangents[n - 1] = m[n - 2];
    for (let i = 1; i < n - 1; ++i) {
        tangents[i] = (m[i - 1] + m[i]) / 2;
    }

    for (let i = 0; i < n - 1; ++i) {
        if (m[i] === 0) {
            tangents[i] = 0;
            tangents[i + 1] = 0;
        } else {
            const alpha = tangents[i] / m[i];
            const beta = tangents[i + 1] / m[i];
            if (alpha < 0) tangents[i] = 0;
            if (beta < 0) tangents[i + 1] = 0;
            const distSq = alpha * alpha + beta * beta;
            if (distSq > 9) {
                const tau = 3 / Math.sqrt(distSq);
                tangents[i] = tau * alpha * m[i];
                tangents[i + 1] = tau * beta * m[i];
            }
        }
    }

    const controlPoints = [];
    for (let i = 0; i < n - 1; ++i) {
        const segmentDx = dx[i];
        controlPoints.push({
            cp1x: points[i].x + segmentDx / 3,
            cp1y: points[i].y + tangents[i] * segmentDx / 3,
            cp2x: points[i + 1].x - segmentDx / 3,
            cp2y: points[i + 1].y - tangents[i + 1] * segmentDx / 3
        });
    }
    return controlPoints;
}
```

```javascript
function formatTooltipMetric(metricKey, slot) {
    if (!slot) return "--";
    if (root.isTimeline25 && slot.isHistoricalUnavailable) return "Past data unavailable";

    switch (metricKey) {
        case "temp": {
            const t = (slot.thermal && slot.thermal.tempC !== null) ? slot.thermal.tempC : (slot.tempC || "--");
            const f = (slot.thermal && slot.thermal.feelsLikeC !== null) ? slot.thermal.feelsLikeC : (slot.FeelsLikeC || t);
            const prob = slot.moisture ? slot.moisture.rainProb : (slot.chanceofrain || 0);
            const mm = slot.moisture ? slot.moisture.precipMM : (slot.precipMM || 0.0);
            return `${t}°C (Feels ${f}°C) • Rain ${prob}% (${mm}mm)`;
        }
        case "rain": {
            const prob = slot.moisture ? slot.moisture.rainProb : (slot.chanceofrain || 0);
            const mm = slot.moisture ? slot.moisture.precipMM : (slot.precipMM || 0.0);
            const cloud = slot.moisture ? slot.moisture.cloudCover : (slot.cloudcover || 0);
            return `Rain ${prob}% • Precip ${mm}mm • Cloud ${cloud}%`;
        }
        case "wind": {
            const spd = slot.wind ? slot.wind.speedKmph : (slot.windspeedKmph || 0);
            const gst = slot.wind ? slot.wind.gustKmph : (slot.WindGustKmph || "--");
            const dir = slot.wind ? slot.wind.dir16 : (slot.winddir16Point || "N");
            const deg = slot.wind ? slot.wind.degree : (slot.winddirDegree || 0);
            return `${spd} km/h • Gust ${gst} km/h • ${dir} (${deg}°)`;
        }
        case "aqi": {
            const epa = slot.aqi ? slot.aqi.epaIndex : 0;
            const cat = WeatherGlyphs.getAqiCategory(epa);
            const p25 = slot.aqi ? slot.aqi.pm2_5 : 0.0;
            const p10 = slot.aqi ? slot.aqi.pm10 : 0.0;
            return `EPA ${epa} (${cat}) • PM2.5: ${p25} µg/m³ • PM10: ${p10} µg/m³`;
        }
        case "baro": {
            const press = slot.atmosphere ? slot.atmosphere.pressureHpa : (slot.pressure || 1013);
            const vis = slot.atmosphere ? slot.atmosphere.visibilityKm : (slot.visibility || 10);
            return `${press} hPa • Vis ${vis} km`;
        }
        case "uv": {
            const uv = slot.solar ? slot.solar.uvIndex : (slot.uvIndex || 0);
            const tier = (uv <= 2) ? "Low" : (uv <= 5 ? "Moderate" : (uv <= 7 ? "High" : (uv <= 10 ? "Very High" : "Extreme")));
            const rad = slot.solar ? slot.solar.shortRad : 0.0;
            return `UV ${uv} (${tier}) • Solar ${rad} W/m²`;
        }
        default:
            return "--";
    }
}
```

##### Temporal Label Formatter (D-54.3-20)
```javascript
function formatTemporalLabel(slotIndex, slot) {
    if (!slot) return "--:--";
    if (root.isTimeline25) {
        const rel = slotIndex - 12;
        if (rel === 0) return `NOW (${slot.timeStr || "12:00"})`;
        if (rel < 0) {
            if (slot.dayOffset < 0) return `Yesterday ${slot.timeStr || ""} (${rel}h)`;
            return `${rel}h (${slot.timeStr || ""})`;
        } else {
            if (slot.dayOffset > 0) return `Tomorrow ${slot.timeStr || ""} (+${rel}h)`;
            return `+${rel}h (${slot.timeStr || ""})`;
        }
    }
    const rawTime = String(slot.time ?? "0");
    const hr = rawTime.length > 2 ? rawTime.slice(0, -2) : (rawTime === "0" ? "00" : rawTime);
    return hr.padStart(2, "0") + ":00";
}
```

## Shared Patterns

### Pattern 1: Zero-Repaint Quiescent Performance Policy
- **Problem:** Hover scrubbing at 60–120 FPS causes heavy CPU and GPU spikes if Canvas 2D `requestPaint()` is invoked on mouse movement.
- **Solution:** Canvas 2D renders only once upon data model change, popup opening, or tab button clicks. Hover scrubbing updates declarative QML Item coordinates (`scrubHairline.x`, `snapDot.x/y`, `tooltipPill.x/y`) via `MouseArea.onPositionChanged`, producing 0 Canvas repaints and maintaining quiescent CPU $\le 1.68\%$.
- **Applied In:** `WeatherGraph.qml` (Scrub overlay & Canvas event handling).

### Pattern 2: Symmetrical Single-Source-of-Truth Y-Scaling (`getMetricY`)
- **Problem:** Writing separate Y coordinate formulas in `Canvas.onPaint` and `MouseArea.onPositionChanged` creates rounding drift and snap dot misalignment.
- **Solution:** A single helper `getMetricY(metricKey, slotIndex, plotTop, plotBottom)` calculates Y coordinates for both painting splines and positioning the hover snap dot, guaranteeing 0px divergence.
- **Applied In:** `WeatherGraph.qml` (`getMetricY()` helper).

### Pattern 3: Fritsch-Carlson Monotone Cubic Spline Interpolation
- **Problem:** Standard Catmull-Rom or cubic Bezier curves produce unnatural oscillations or undershoot below 0 for bounded metrics like rain probability, wind speed, or UV index.
- **Solution:** Fritsch-Carlson algorithm computes tangents based on secant slopes and scales them whenever $\alpha^2 + \beta^2 > 9$, preserving strict monotonicity and preventing local overshoot.
- **Applied In:** `WeatherGraph.qml` (`computeMonotoneSplineControlPoints()`).

### Pattern 4: Smart Tooltip Clamping & Vertical Flip
- **Problem:** The hover tooltip pill clips past card edges when hovering over slot 0 (-12h) or slot 24 (+12h), and collides with top tab pills when hovering near the top of the curve ($Y < 60\text{px}$).
- **Solution:** Dual-axis positioning clamping:
  - Horizontal: `Math.max(6, Math.min(parent.width - implicitWidth - 6, root.activeScrubX - implicitWidth / 2))`
  - Vertical: `(root.activeScrubY < 60) ? (root.activeScrubY + 14) : (root.activeScrubY - implicitHeight - 8)`
- **Applied In:** `WeatherGraph.qml` (`tooltipPill` position bindings).

### Pattern 5: Headless Node.js Test Harness Pattern
- **Problem:** Running tests against QML components usually requires launching a Wayland/quickshell GUI daemon, which violates CI/headless test environments.
- **Solution:** Extract JavaScript helper functions (`getMetricY`, `computeMonotoneSplineControlPoints`) from the QML source using regex and evaluate them headlessly in Node.js via `new Function(...)`.
- **Applied In:** `scripts/phase54.3-weather-assert.sh`.
