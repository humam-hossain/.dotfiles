```python
def resolve_history_rollover(prev_cache: dict | None, new_wwo_data: dict) -> list:
    """
    Computes top-level history array based on calendar date difference.
    0-cost promotion of previous day's weather[0] upon midnight rollover.
    Invalidates history (returns []) if offline for >1 calendar day.
    """
    if not prev_cache:
        return []

    new_weather = new_wwo_data.get("weather", [])
    if not new_weather:
        return []
    new_date_str = new_weather[0].get("date")

    prev_data = prev_cache.get("data") or {}
    prev_weather = prev_data.get("weather", [])
    prev_weather_0 = prev_weather[0] if prev_weather else None
    prev_date_str = prev_weather_0.get("date") if prev_weather_0 else None
    prev_history = prev_cache.get("history", [])

    if not new_date_str:
        return []

    try:
        new_d = datetime.strptime(new_date_str, "%Y-%m-%d").date()
        yesterday_str = (new_d - timedelta(days=1)).strftime("%Y-%m-%d")

        if prev_date_str:
            prev_d = datetime.strptime(prev_date_str, "%Y-%m-%d").date()
            day_diff = (new_d - prev_d).days

            if day_diff == 1:
                # Midnight rollover: promote yesterday's day object into history (0 API cost)
                return [prev_weather_0]
            elif day_diff == 0:
                # Same day: retain valid yesterday entry from existing history
                return [h for h in prev_history if h.get("date") == yesterday_str]
            else:
                # Multi-day downtime (>1) or clock anomaly (<0): invalidate cleanly
                return []
        elif prev_history:
            # Fallback check on existing history
            return [h for h in prev_history if h.get("date") == yesterday_str]
    except Exception:
        return []

    return []
```

```qml
        if (root._seriesCache[metricPath]) return root._seriesCache[metricPath];

        const parts = metricPath.split(".");
        const series = new Array(25);

        for (let i = 0; i < 25; ++i) {
            const slot = root.timeline[i];
            let val = null;
            if (parts.length === 2) {
                val = slot[parts[0]]?.[parts[1]];
            } else if (parts.length === 1) {
                val = slot[metricPath] ?? slot.thermal?.[metricPath] ?? slot.moisture?.[metricPath];
            }
            series[i] = (val !== null && val !== undefined && !isNaN(val)) ? Number(val) : (parts[0] === "thermal" ? null : 0);
        }

        root._seriesCache[metricPath] = series;
        return series;
    }
```
